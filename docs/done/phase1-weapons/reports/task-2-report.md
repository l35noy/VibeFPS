### `scripts/hurtbox.gd` (43 строки)

```gdscript
class_name Hurtbox
extends Area3D

## Хитбокс: зона, в которую можно попасть.
##
## Здоровья здесь нет и не будет. Хитбокс только сообщает о попадании
## (`damage_taken`), а состояние здоровья живёт у того, кто этот хитбокс
## владеет — в Фазе 2 это враг. Если бы урон уменьшал поле внутри хитбокса,
## у каждого поражённого места появилось бы своё здоровье, и «у врага 100 HP»
## перестало бы быть правдой.

signal damage_taken(amount: float)

## Множитель урона. Обычный враг — 1.0. В Фазе 2 через это поле делается
## слабое место босса с ×2, поэтому контракт (умножение при нанесении, а не при
## поглощении) должен быть верным уже сейчас, даже пока его никто не читает.
@export var damage_multiplier: float = 1.0

## Слой 4 — `enemy_hurtbox`, как назван слой в project.godot.
const _COLLISION_LAYER := 1 << 3
## Маска 1 — мировая геометрия. Хитбокс не должен ловить игрока: ни игрок,
## ни его хитбокс в маске не стоят, поэтому выстрел, прошедший мимо цели,
## не может зацепить игрока и его не перебить.
const _COLLISION_MASK := 1 << 0


## Слой и маска задаются здесь, а не в сцене: правило «кого я ловлю» и логика
## попадания лежат в одном файле, и сцена не может их разъехать.
func _ready() -> void:
	collision_layer = _COLLISION_LAYER
	collision_mask = _COLLISION_MASK
	# Хитбоксу нечего находить самому — он цель, а не наблюдатель, — но его
	# собственные пересечения не нужны и только тратят время на запросы.
	monitoring = false
	# А вот обнаруживаемым он обязан быть: без этого снаряд, который его ищет,
	# его не увидит.
	monitorable = true


## Сообщить попадание. Ничего не меняет в собственном состоянии: ни здоровья,
## ни «уже полученного урона» здесь нет.
func apply_damage(amount: float) -> void:
	damage_taken.emit(amount * damage_multiplier)
```

`apply_damage` дословно из брифа. Слой/маска — в `_ready()`, как требовал бриф.
`monitoring = false` / `monitorable = true` — решение, о котором ниже (§7).

### `scripts/projectile.gd` (173 строки)

```gdscript
class_name Projectile
extends Area3D

## Снаряд: летит по прямой с конечной постоянной скоростью, попадает один раз
## и исчезает.
##
## Три решения здесь — игровые правила, а не технические детали:
##
## 1. Конечная скорость. На 40 м/с снаряд пистолета пролетает 15 м за 0.375 с.
##    Это окно, за которое цель успевает отойти, поэтому и игрок, и враг могут
##    уклоняться. Мгновенный (хитскан) выстрел это окно убирает, и уклонение
##    становится невозможным — значит, скорость обязана быть в данных, а не
##    бесконечной.
## 2. Никакого ускорения и отскока: направление нормализуется один раз при
##    запуске, дальше скорость постоянна. Сглаживание, интерполяция или
##    «доводка» до цели сломали бы правило «сколько метров за секунду» и,
##    вместе с ним, окно уклонения.
## 3. Один выстрел — одно попадание. Снаряд, переживший попадание, за тот же
##    кадр достал бы вторую цель и нанёс урон двоим.

## Выстрел попал во что-то. Для диагностики и для проверок: попадание, которое
## нельзя увидеть, невозможно и отладить.
signal hit_something(target: Node)

## Скорость в м/с. Конечная: см. пункт 1 выше.
@export var speed: float = 40.0
## Сколько урона наносит одно попадание. Хитбокс применяет свой
## `damage_multiplier` — множитель живёт у цели, а не у снаряда.
@export var damage: float = 30.0
## Через сколько секунд снаряд истекает, если ни во что не попал. Страховка от
## снаряда, улетевшего в пустоту на шум в приложении.
@export var lifetime: float = 3.0

## Слой 5 — `projectile`.
const _COLLISION_LAYER := 1 << 4
## Маска 1 | 4: мировая геометрия (слой 1) и хитбоксы врага (слой 4). Слоя
## хитбокса игрока (7) в маске нет намеренно: игрок не должен получать урон от
## собственных выстрелов, прошедших мимо цели.
const _COLLISION_MASK := (1 << 0) | (1 << 3)

## Прозрачность шлейфа в начале полёта и её потолок.
const _TRAIL_PEAK_ALPHA := 0.6
const _TRAIL_COLOR := Color(0.15, 0.15, 0.15)

## Направление нормализовано один раз здесь; дальше оно только читается.
var _direction := Vector3.FORWARD
var _age := 0.0
## Снаряд кончился: пересечения больше не обрабатываются, даже если движок
## успеет прислать их в том же проходе.
var _spent := false
var _trail: MeshInstance3D
var _trail_material: StandardMaterial3D


## Слой и маска задаются здесь, а не в сцене: правило «кого я вижу» и логика
## попадания лежат в одном файле, и сцена не может их разъехать.
func _ready() -> void:
	collision_layer = _COLLISION_LAYER
	collision_mask = _COLLISION_MASK
	# Снаряду нужно видеть другие Area3D, но сам он в чужие пересечения попадать
	# не должен: иначе чужой код, слушающий пересечения, увидит каждый снаряд.
	monitoring = true
	monitorable = false
	area_entered.connect(_on_area_entered)
	body_entered.connect(_on_body_entered)
	_setup_trail()
	# До launch() снаряд стоит на месте: выстрел сам решает, когда снаряд
	# появится в мире.
	set_physics_process(false)


## Задать направление и включить полёт. Узел при этом не двигается: решение
## «откуда и куда» принимает выстрел, а снаряд начинает оттуда, где уже стоит.
func launch(direction: Vector3) -> void:
	if direction.is_zero_approx():
		# Нулевое направление — это не «лети куда попало», а отсутствие выстрела:
		# нормализация нуля дала бы NaN, и снаряд улетел бы в никуда молча.
		return
	_direction = direction.normalized()
	_age = 0.0
	_spent = false
	set_physics_process(true)


func _physics_process(delta: float) -> void:
	# Константа скорости: ровно speed * delta каждый тик, без ускорения, без
	# сглаживания. Расстояние за секунду — ровно speed, и это проверяемо.
	global_position += _direction * (speed * delta)
	_age += delta
	_fade_trail()
	if _age >= lifetime:
		# Ни во что не попал: снимаем снаряд сцены, чтобы он не ждал своей
		# геометрии вечно.
		queue_free()


## Урон наносится только хитбоксу. Любой другой Area3D — это «снаряд пролетел
## мимо», а не ошибка: вслепую вешать урон на любую зону нельзя, иначе
## триггерные зоны уровня начнут принимать пули.
func _on_area_entered(area: Area3D) -> void:
	if _spent:
		return
	if not (area is Hurtbox):
		return
	(area as Hurtbox).apply_damage(damage)
	hit_something.emit(area)
	_spend()


## Геометрия мира (слой 1): стена, пол, укрытие. Урона не несёт, но снаряд
## кончается — иначе пуля продолжала бы лететь сквозь стену.
func _on_body_entered(body: Node3D) -> void:
	if _spent:
		return
	hit_something.emit(body)
	_spend()


## Снаряд отстрелял своё единственное попадание.
##
## Порядок обязателен. `monitoring` выключается через set_deferred, а не
## присваиванием: этот обработчик вызывается движком в том самом проходе, в
## котором обновляются пересечения, и прямое изменение состояния там даёт
## ошибку Godot «can't change this state while flushing queries». Эта ошибка
## уходит в консоль, а не в проверку, поэтому её приходится предотвращать
## здесь, а не ловить.
func _spend() -> void:
	_spent = true
	set_physics_process(false)
	set_deferred("monitoring", false)
	queue_free()


## Шлейф: дочерний MeshInstance3D, который бледнеет по времени жизни.
##
## Сцена может задать геометрию и цвет шлейфа сама (Task 5 задаёт их из данных
## оружия); если её нет, снаряд остаётся рабочим самодостаточным. Затухание —
## уменьшение `albedo_color.a` каждый кадр, а НЕ Tween: Tween продолжает
## работать и после попадания, а шлейф обязан умирать вместе со снарядом, и
## иначе в месте попадания оставался бы висящий в воздухе полупрозрачный шар.
func _setup_trail() -> void:
	if _trail == null:
		for child in get_children():
			if child is MeshInstance3D:
				_trail = child
				break
	if _trail == null:
		_trail = MeshInstance3D.new()
		_trail.name = "Trail"
		var mesh := SphereMesh.new()
		mesh.radius = 0.08
		mesh.height = 0.16
		_trail.mesh = mesh
		add_child(_trail)
	var mat := _trail.material_override as StandardMaterial3D
	if mat == null:
		mat = StandardMaterial3D.new()
		_trail.material_override = mat
		var color := _TRAIL_COLOR
		color.a = _TRAIL_PEAK_ALPHA
		mat.albedo_color = color
	mat.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	if is_zero_approx(mat.albedo_color.a):
		mat.albedo_color.a = _TRAIL_PEAK_ALPHA
	_trail_material = mat


func _fade_trail() -> void:
	if _trail_material == null or lifetime <= 0.0:
		return
	var color := _trail_material.albedo_color
	color.a = _TRAIL_PEAK_ALPHA * clampf(1.0 - _age / lifetime, 0.0, 1.0)
	_trail_material.albedo_color = color
```

Поля (`speed`, `damage`, `lifetime`), сигнал `hit_something`, слои и маска — дословно
по брифу. Остальное — реализация правил брифа.

Скрипты под тестом берутся через `load()` по пути, а не через `class_name`.
Это сделано намеренно: статическая ссылка на несуществующий `class_name` убивает
всю проверку на этапе разбора («Identifier not declared in the current scope») и
не сообщает ничего о том, чей контракт нарушен. Через `load()` отсутствующий
файл называется по имени, а остальные утверждения остаются исполнимыми.

## 5. Мутации

### Мутация 1 (обязательная): снять `queue_free()` из обработчика попадания

SHA-256 `scripts/projectile.gd` до: `32D588851DFD34A8D38AA51D2899332E916018D3241D54E1E2B97BEE7DC6E31B`

Бриф ожидал, что упадёт утверждение «урон приходит ровно один раз». Оно **не**
упало, и это не дефект проверки, а следствие того, что в `_spend()` три
независимых предохранителя, а не один. Чтобы это утверждение вообще могло
покраснеть, пришлось снимать их по одному — так и выяснилось, чем именно оно
держится. Лестница мутаций (каждый шаг — отдельный прогон):

| Шаг | Что убрано | Красные утверждения |
|---|---|---|
| 1a | `queue_free()` из `_spend()` | 5: «снаряд исчез у цели», «снаряд снят стеной», «остановлен стеной, а не по времени», «выстрел кончился у ближней цели», «снаряд не кончился на хитбоксе игрока» (все — «gone after -1 ticks», т.е. узел не исчез вовсе) |
| 1b | + `if _spent: return` из обоих обработчиков | 6: те же 5 + «один выстрел сообщает один контакт, хотя в кадре было два» (получено 2) |
| 1c | + `set_physics_process(false)` и `set_deferred("monitoring", …)` (то есть `_spend()` обесточен целиком) | **13**, включая ровно то, что предсказывал бриф: «**дальняя цель не получила ничего** (got 1: [30.0])» и «**один выстрел дал ровно одно событие урона во всём мире** (got 2)» |

Вывод 1c — тот самый механизм, который бриф просил показать: снаряд пережил
попадание, долетел до второй цели и ударил обе. Значит утверждения «урон ровно
один раз» умеют падать, и падают по причине, а не по невезению.

Важная деталь для партнёра: **проверка ловит и 1a**, то есть снятие одного
`queue_free()` тоже красное — просто падает оно на «снаряд исчез», а не на
счётчике урона.

Восстановление (три обратных редактирования) и сверка:

```

прогон: projectile PASS, RESULT: 1 passed, 0 failed, 0 skipped, EXIT=0
```

## 7. Решения, которые бриф оставил открытыми

1. **Кто создаёт форму столкновений снаряда.** Бриф Task 2 требует шлейф
   (дочерний `MeshInstance3D`), но форму столкновений не упоминает; Task 5
   описывает сцену снаряда как `Projectile` + дочерний `MeshInstance3D` — то
   есть **без `CollisionShape3D`**. У `Area3D` без формы нет пересечений
   вообще, и снаряд в игре не попадёт ни во что.
   Решение здесь: скрипт форму не создаёт (в брифе её нет, а Task 5 — её место),
   проверка создаёт сферу радиуса 0.08 — ровно `WeaponData.projectile_radius` —
   себе сама. **Task 5 обязан добавить в `projectile.tscn` `CollisionShape3D`**,
   иначе боевой сцены не будет. Это самое важное, что нужно передать дальше.
2. **Кто задаёт геометрию и цвет шлейфа.** `_setup_trail()` берёт первый
   существующий дочерний `MeshInstance3D`, а если его нет — создаёт сферу сам.
   Так сцена Task 5 может задать геометрию и цвет из данных оружия, а снаряд
   при этом остаётся рабочим без сцены (что и проверяется сейчас).
3. **`monitoring`/`monitorable` у хитбокса.** Бриф задаёт только слой и маску.
   `monitoring = false` (хитбокс — цель, а не наблюдатель) и
   `monitorable = true` (иначе мониторящий снаряд его не увидит). Обе настройки
   зафиксированы утверждениями 16–17.
4. **Пункт 3 Review Focus («узкий хитбокс между отсчётами») закрыт частично, и
   это осознанно.** Обнаружение через сигналы `Area3D` — дискретное: Jolt
   сравнивает позиции в тиках, а не объём движения. Один тик — это ровно 1/3 м
   (40 м/с при 120 Гц), поэтому цель уже 1/3 м не может быть задета надёжно, и
   никакая правка снаряда это не чинит — нужен `ShapeCast3D`/`RayCast3D`, а бриф
   прямо запрещает менять механизм. Что сделано: все мишени шириной 0.5–1.0 м
   (заведомо больше тика), и инвариант «шаг меньше мишени» проверяется
   косвенно — утверждениями 19, 21, 36, 41, 45, 60, где смерть снаряда всегда
   измерена в тиках и всегда раньше контрольного события. Если Фаза 2 поднимет
   скорость так, что тик станет шире мишени, эти проверки начнут падать сами.
   Настоящее непрерывное попадание для тонких хитбоксов — вопрос Фазы 2, и его
   стоит записать туда.
5. **Пункт 5 Review Focus («два снаряда в один хитбокс в одном кадре») закрыт на
   уровне снаряда, а не хитбокса.** Снаряд гарантирует, что одно попадание
   даёт одно событие урона даже тогда, когда в одном кадре приходит и
   `area_entered`, и `body_entered` (утверждения 61–65). Два *разных* снаряда
   в один хитбокс дадут два события — и это правильно: решение «применить ли
   второй урон» принадлежит владельцу здоровья, то есть врагу Фазы 2, а не
   хитбоксу, который по брифу не хранит состояния.
6. **Нулевое направление в `launch()`** считается «выстрела нет»: нормализация
   нуля дала бы `NaN`, и снаряд улетел бы в никуда молча. Проверка этого не
   касается (это защита, а не контракт); если партнёр считает иначе — легко
   переделать.
7. **Проверка не использует `class_name` статически** (см. §3) — сознательный
   размен типизации на диагностируемость падения.

## 8. Дефекты, найденные и исправленные по ходу

1. **Шлейф стартовал с непрозрачного `a = 1.0`** вместо потолка 0.6: цвет
   `_TRAIL_COLOR` задан без альфы, и проверка «начинается видимым» проходила
   случайно. Исправлено: альфа выставляется явно при создании материала.
2. **Утверждение «полёт не удовлетворяется телепортом» было написано неверно**
   (`travelled < _WALL_FRONT` при `travelled == 20.0 == _WALL_FRONT` — стены в
   том мире вообще нет). Заменено на измеряемое «первые 10 м пройдены примерно
   за 30 тиков» с допуском в один тик: 30 шагов по 1/3 м дают 9.999999999999998,
   то есть ровно на волосок меньше 10 — допуск задокументирован прямо в тексте
   утверждения.
3. **Потерянный счётчик кадров в проверке `lifetime`**: `_frames_until_free`
   считает от момента вызова, а 15 тиков уже были прожиты, поэтому снаряд с
   lifetime 30 тиков «умирал на 16-м» и валил проверку. Переписано на один
   цикл со счётчиком от запуска.
4. **Имена узлов в проверке совпадали**, и `compare`-сообщения показывали
   `@Area3D@31` вместо имени. Хитбоксы теперь получают уникальные имена.
5. **Проверка на отсутствие твина искала слово «Tween»** и попадала на
   собственное объяснение в комментарии скрипта. Заменена на поиск API-имён.
